%    - logo / 标题 / 作者 / 联系方式整体靠左并上移
%    - 可选摘要：浅灰（#F7F7F7）圆角底框，高度随文字量自适应
%    - 未设置 \reportabstract 时不显示摘要

\thispagestyle{empty}
\begingroup
\topskip 0pt%                                                 % 消除页面顶部自动间距
\parskip 0pt%                                                 % 各元素间距由下方 \vspace 控制
\noindent\vbox to\textheight{%
    \vfil%                                                    % 上方留白（少于下方，整体上移）
    {\raggedright
        % --- Logo（靠左）---
        \ifx\@reportlogo\undefined\else
            \IfFileExists{\@reportlogo}{%
                \includegraphics[height=2cm]{\@reportlogo}\par
            }{}%
        \fi
        \vspace{1.2em}

        % --- 报告标题 ---
        {\Huge\bfseries \@reporttitle \par}
        \vspace{1em}

        % --- 作者 ---
        {\large \@reportauthor \par}
        \vspace{0.8em}

        % --- 联系方式 ---
        {\normalsize \@reportcontact \par}
        \par
    }%
    % --- 可选摘要（无则不显示）---
    \ifx\@reportabstract\undefined\else
        \vspace{2em}%
        \begin{tcolorbox}[
            enhanced,
            breakable,
            colback=abstractfill,
            colframe=abstractfill,
            boxrule=0pt,
            arc=10pt,
            left=20pt,
            right=20pt,
            top=20pt,
            bottom=20pt,
        ]
        \@reportabstract
        \end{tcolorbox}%
    \fi
    \vfil%                                                    % 下方留白更多（2 倍于上方）
    \vfil%
}
\endgroup
